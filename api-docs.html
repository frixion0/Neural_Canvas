Image Generation Example

You can use our proxy endpoint to generate images with various models, including 'flux', 'uncen', 'img3', and 'img4'. Simply change the model in the request body.

async function generateImage(model, prompt) {
  const response = await fetch('https://9000-firebase-studio-1754744124282.cluster-nzwlpk54dvagsxetkvxzbvslyi.cloudworkstations.dev/api/proxy', {
    method: 'POST',
    headers: { 
      'Content-Type': 'application/json' 
    },
    body: JSON.stringify({
      model: model, // e.g., 'img3', 'img4', 'flux'
      prompt: prompt,
    }),
  });
  const data = await response.json();
  if (response.ok) {
    console.log('Image URL:', data.imageUrl);
    return data.imageUrl;
  } else {
    console.error('Error:', data.error);
    throw new Error(data.error);
  }
}
generateImage('img3', 'A photorealistic image of a majestic lion in the savanna at sunset');

Available Models:

  • `qwen`
  • `uncen`
  • `flux`
  • `fluxpro`
  • `sana`
  • `sana-flash`
  • `img3` (Imagen 3 - Default)
  • `img4` (Imagen 4)

Image Editing API

Edit images using natural language instructions with our AI image manipulation endpoint.

async function editImage(imageDataUri, instructions) {
  const response = await fetch('https://9000-firebase-studio-1755789802422.cluster-73qgvk7hjjadkrjeyexca5ivva.cloudworkstations.dev/api/manipulate', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      photoDataUri: imageDataUri,
      instructions: instructions
    }),
  });
  const data = await response.json();
  if (response.ok) {
    console.log('Edited Image URL:', data.editedPhotoDataUri);
    return data.editedPhotoDataUri;
  } else {
    console.error('Error:', data.error);
    throw new Error(data.error);
  }
}

// Example usage
const base64Image = 'data:image/jpeg;base64,...';
editImage(base64Image, 'Make the sky blue and add a rainbow');
